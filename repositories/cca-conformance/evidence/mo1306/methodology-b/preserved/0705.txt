def now():return datetime.now(timezone.utc).isoformat()
def sha(data):return 'sha256:'+hashlib.sha256(data).hexdigest()
def row(path):
 data=path.read_bytes()
 return {'path':path.relative_to(ROOT).as_posix(),'byteLength':len(data),'sha256':sha(data)}
def put(path,value):
 path.parent.mkdir(parents=True,exist_ok=True)
 assert not path.exists(),str(path)
 path.write_text(json.dumps(value,sort_keys=True,separators=(',',':'))+'\n',encoding='utf-8')
 return row(path)

class SameGitHubAPI(urllib.request.HTTPRedirectHandler):
 def redirect_request(self,req,fp,code,msg,headers,newurl):
  parsed=urllib.parse.urlsplit(newurl)
  if parsed.scheme!='https' or parsed.netloc!='api.github.com':
   raise RuntimeError('Unexpected API redirect')
  return super().redirect_request(req,fp,code,msg,headers,newurl)

def credential():
 env={**os.environ,'GIT_TERMINAL_PROMPT':'0','GCM_INTERACTIVE':'Never','GIT_OPTIONAL_LOCKS':'0'}
 result=subprocess.run(['git','credential','fill'],cwd=ROOT,env=env,
  input=b'protocol=https\nhost=github.com\n\n',capture_output=True,timeout=30,creationflags=subprocess.CREATE_NO_WINDOW)
 assert result.returncode==0,'Credential helper unavailable; secret output not retained'
 values=dict(line.split('=',1) for line in result.stdout.decode().splitlines() if '=' in line)
 assert values.get('password'),'Credential unavailable'
 return values['password']

def get(path,token):
 assert not path.startswith('/') and '..' not in path.split('/')
 headers={'User-Agent':'MO1306-Phase3AR-Resolution-ReadOnly','Accept':'application/vnd.github+json',
          'X-GitHub-Api-Version':API_VERSION,'Authorization':'Bearer '+token}
 request=urllib.request.Request('https://api.github.com/'+path,headers=headers,method='GET')
 start=now();opener=urllib.request.build_opener(SameGitHubAPI())
 try:
  with opener.open(request,timeout=30) as response:
   status=response.status;raw=response.read(4194305);url=response.url;received=response.headers
 except urllib.error.HTTPError as error:
  status=error.code;raw=error.read(4194305);url=error.url;received=error.headers
 assert len(raw)<=4194304,'API response too large'
 body=json.loads(raw) if raw else None
 if isinstance(body,dict) and 'temp_clone_token' in body:body['temp_clone_token']=None
 selected={name:received.get(name) for name in ['date','x-github-request-id','x-oauth-scopes',
  'x-accepted-oauth-scopes','x-accepted-github-permissions','github-authentication-token-expiration'] if received.get(name)}
 return {'method':'GET','path':path,'status':status,'url':url,'startedUtc':start,'endedUtc':now(),
         'apiVersion':API_VERSION,'responseHeaders':selected,'body':body}
