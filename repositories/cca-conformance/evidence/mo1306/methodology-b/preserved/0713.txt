function describe(args){return args.map(v=>typeof v==='bigint'?String(v):Buffer.isBuffer(v)||ArrayBuffer.isView(v)?{type:v.constructor.name,byteLength:v.byteLength}:v instanceof URL?v.href:v instanceof Error?errorRecord(v):typeof v==='function'?{type:'function'}:v);}
function resultRecord(v){if(v&&typeof v.isFile==='function')return {isFile:v.isFile(),isDirectory:v.isDirectory(),isSymbolicLink:v.isSymbolicLink(),size:String(v.size),nlink:String(v.nlink),mode:Number(v.mode)};if(Buffer.isBuffer(v))return {byteLength:v.length};if(Array.isArray(v))return {count:v.length};return v;}
for(const method of methods){

  counts[method]=(counts[method]??0)+1;
  const trackingBegin=now();
  const operands=(method==='renameSync'?args.slice(0,2):args.slice(0,1)).map(v=>typeof v==='number'?fdPaths.get(v)??null:pathValue(v));

   if(tracked){const tick=now();log('fsAfter',{method,operands,elapsedMs:after-before,result:resultRecord(value)});fsObservationOverheadMs+=now()-tick;}
   return value;
  }catch(error){
   const tick=now();log('fsError',{method,operands,arguments:describe(args),elapsedMs:tick-before,error:errorRecord(error),workspaceScoped:tracked});fsObservationOverheadMs+=now()-tick;throw error;
  }
 };
}
cp.spawn=function(...args){

 child.stderr.emit=function(event,...rest){if(event==='data')log('helperStderr',{ordinal,pid:child.pid,byteLength:rest[0].length,text:rest[0].subarray(0,2048).toString('utf8')});return Reflect.apply(errEmit,this,[event,...rest]);};
 return child;
};

for(const name of methods){if(Object.hasOwn(originals[name],'native'))Object.defineProperty(fs[name],'native',Object.getOwnPropertyDescriptor(originals[name],'native'));}
