def j(value):
    return (json.dumps(value, sort_keys=True, separators=(',', ':'), ensure_ascii=False) + '\n').encode('utf8')


def sha(data):
    return 'sha256:' + hashlib.sha256(data).hexdigest()


def ref(path):
    raw = path.read_bytes()
    return {'path': path.relative_to(ROOT).as_posix() if path.is_relative_to(ROOT) else str(path), 'byteLength': len(raw), 'sha256': sha(raw)}


def put(path, value):
    raw = j(value)
    assert len(raw) <= JSON_CAP, ('ENGINEERING_JSON_CAP', str(path), len(raw))
    path.parent.mkdir(parents=True, exist_ok=True)
    # Run-local files are immutable after creation.
    with path.open('xb') as stream:
        stream.write(raw)
    return ref(path)


def inventory(package):
    return [ref(p) | {'path': p.relative_to(package).as_posix()}
            for p in sorted(package.rglob('*')) if p.is_file()]


def compact(record):
    result = copy.deepcopy(record)
    dictionary = {}
    rows = result.get('lifetimes', []) + [p for s in result.get('observation', {}).get('samples', []) for p in s['processes']]
    for entry in rows:
        if isinstance(entry.get('commandLine'), str):
            raw = entry['commandLine']; key = sha(raw.encode('utf8'))
            dictionary[key] = raw; entry['commandLine'] = {'sha256': key}
    result['traceEncoding'] = {'kind': 'LosslessCommandLineDictionary', 'version': '1.0.0', 'commandLines': dictionary}
    return result

    def raw_logs(self, name, record):
        rows = []
        for field in ('stdout', 'stderr'):
            value = record.get(field, '')
            raw = value if isinstance(value, bytes) else value.encode('utf8')
            assert len(raw) <= JSON_CAP, 'RAW_LOG_CAP'
            p = self.output / 'logs' / (name + '.' + field + '.txt')
            p.parent.mkdir(parents=True, exist_ok=True)
            with p.open('xb') as stream: stream.write(raw)
            rows.append(ref(p))
        return rows

    def observed(self, name, args, expected=0, package=INSTALLED, timeout=80):
        self.budget()
        argv = list(map(str, args))
        result = None
        partial = []
        original_snapshot = observer.snapshot
        def capture(*a, **kw):
            rows = original_snapshot(*a, **kw)
            partial.append({'elapsedMs': round((time.monotonic() - begun) * 1000), 'processes': copy.deepcopy(rows), 'totalRssBytes': sum(r['rssBytes'] for r in rows)})
            return rows
        begun = time.monotonic()
        observer.snapshot = capture
        failure = None
        try:
            result = observer.execute(argv, INSTALLED.parent, env=self.env, timeout=timeout, interval=.1)
            assert result['exitCode'] == expected, ('EXIT_MISMATCH', name, expected, result['exitCode'])
            assert result['lifetimes'], 'NO_OBSERVED_PROCESSES'
            roots = [r for r in result['lifetimes'] if r['pid'] == result['rootPid']]
            assert len(roots) == 1 and roots[0]['commandLine'] == subprocess.list2cmdline(argv), 'ROOT_ARGV_IDENTITY'
            result['topologyValidation'] = validate_trace(result, self.ids, package)
            assert result['cleanup']['elapsedMs'] <= 2000 and not result['cleanup']['remainingPids'], 'CLEANUP_BOUND'
            assert result['elapsedMs'] <= (75000 if timeout == 80 else timeout * 1000), 'WALL_CLOCK_LIMIT'
            result['status'] = 'PASS'
            return result
        except BaseException as error:
            failure = type(error).__name__ + ': ' + str(error)
            if result is None:
                lifetimes = {}
                for sample in partial:
                    for p in sample['processes']: lifetimes[(p['pid'], p['creationTime100ns'])] = p
                remaining = []
                for (pid, born), p in lifetimes.items():
                    current = observer.details(pid)
                    if current and current['creationTime100ns'] == born: remaining.append({'pid': pid, 'creationTime100ns': born})
                result = {'command': argv, 'elapsedMs': round((time.monotonic() - begun) * 1000), 'lifetimes': list(lifetimes.values()), 'observation': {'samples': partial}, 'cleanup': {'remainingIdentities': remaining}, 'stdout': '', 'stderr': '', 'limitation': 'Observer failed before returning complete streams; partial samples and remaining observed identities retained.'}
            result.update(status='FAIL', validationError=failure)
            raise
        finally:
            observer.snapshot = original_snapshot
            result.update(id=name, expectedExitCode=expected, bindings=self.bindings)
            result['rawLogs'] = self.raw_logs(name, result)
            reference = put(self.output / 'traces' / (name + '.json'), compact(result))
            self.commands.append({'id': name, 'status': result['status'], 'trace': reference})
            print(name, result['status'], result.get('elapsedMs'), 'ms', flush=True)
