from datetime import datetime,timezone
import hashlib,json,os,subprocess,sys,urllib.request,urllib.error
ROOT=Path(__file__).resolve().parents[4];OUT=ROOT/'repositories/cca-conformance/evidence/mo1306/phase3ar';CACHE=ROOT/'.cache/mo1306-phase3ar'
C3AB='9e1bbff99c23cdbc4c80f11e87eadb89a36e8f44';BRANCH='mo1306-certification';REPO='moelsaka01/memoryos-specification'

def now():return datetime.now(timezone.utc).isoformat()
def git(*args,env=None,input=None,check=True):
 r=subprocess.run(['git',*args],cwd=ROOT,env=env or ENV,input=input,capture_output=True,timeout=180,creationflags=subprocess.CREATE_NO_WINDOW)
 if check:assert r.returncode==0,(args,r.returncode,r.stderr.decode('utf8','replace'))
 return r

def put(name,obj):
 p=OUT/name;p.parent.mkdir(parents=True,exist_ok=True);assert not p.exists(),p;p.write_text(json.dumps(obj,sort_keys=True,separators=(',',':'))+'\n',encoding='utf8')
def refs():return git('ls-remote','--heads','--tags','origin').stdout.decode()
def api(path,method='GET',data=None,auth=False):
 headers={'User-Agent':'MO1306-Phase3AR-Certification','Accept':'application/vnd.github+json','X-GitHub-Api-Version':'2022-11-28'}
 if auth:
  r=git('credential','fill',input=b'protocol=https\nhost=github.com\n\n',check=False)
  assert r.returncode==0,'Git credential helper unavailable; no secret output retained'
  credential=dict(line.split('=',1) for line in r.stdout.decode().splitlines() if '=' in line)
  assert credential.get('password'),'GitHub credential unavailable';headers['Authorization']='Bearer '+credential['password']
 body=None if data is None else json.dumps(data).encode();req=urllib.request.Request('https://api.github.com/'+path,data=body,headers=headers,method=method);start=now()
 try:
  with urllib.request.urlopen(req,timeout=35) as r:status=r.status;raw=r.read(4*1024*1024);url=r.url
 except urllib.error.HTTPError as e:status=e.code;raw=e.read(4*1024*1024);url=e.url
 try:value=json.loads(raw)
 except ValueError:value=raw.decode('utf8','replace')
 return dict(method=method,path=path,url=url,status=status,startedUtc=start,endedUtc=now(),body=value)
