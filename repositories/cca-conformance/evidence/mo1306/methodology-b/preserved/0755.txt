function snapshot(directory) {
 if(!fs.existsSync(directory))return [];
 return fs.readdirSync(directory,{withFileTypes:true}).flatMap(entry=>{const full=path.join(directory,entry.name);return entry.isDirectory()?[{path:rel(full),directory:true},...snapshot(full)]:[row(full)];}).sort((a,b)=>a.path.localeCompare(b.path));
}

}

async function rejectsPublication(directory,options,expected) {

 return {rejectionCode:seen.code};
}
function watchStaged(out,controller) {
 let observation=null;
 const timer=setInterval(()=>{
  if(observation||!fs.existsSync(out))return;
  for(const entry of fs.readdirSync(out,{withFileTypes:true})) {
   if(!entry.isDirectory())continue;
   const directory=path.join(out,entry.name),pending=path.join(directory,'.pending');
   if(!fs.existsSync(pending))continue;
   const files=fs.readdirSync(pending).filter(name=>fs.lstatSync(path.join(pending,name)).isFile());
   if(files.length) {
    observation={runId:entry.name,directory:rel(directory),stagedFiles:files.sort(),observedElapsedMs:Math.round(performance.now()-started),markerAbsentAtAbort:!fs.existsSync(path.join(directory,'memoryos-ci-complete.json'))};
    clearInterval(timer);controller.abort();return;
   }
  }
 },5);
 return {stop:()=>clearInterval(timer),read:()=>observation};
}
