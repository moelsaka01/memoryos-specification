def observed(name,args,expected=0,env=None,package=PKG,timeout=80):
 record=execute(list(map(str,args)),ROOT,env=env or ENV,timeout=timeout)
 record.update(id=name,expectedExitCode=expected,distributionDigest=DISTRIBUTION,harnessSha256=sha(Path(__file__).read_bytes()),status='FAIL')
 try:
  assert record['exitCode']==expected,(name,record['stdout'],record['stderr'])
  assert record['lifetimes'] and any(p['pid']==record['rootPid'] for p in record['lifetimes']),'observer saw no supervisor'
  record['topologyValidation']=validate_trace(record,IDS,package)
  record['status']='PASS'
 finally:
  # Dictionary encoding is lossless and keeps bounded records below 2 MiB.
  original=copy.deepcopy(record);dictionary={}
  for p in record['lifetimes']+[p for sample in record['observation']['samples'] for p in sample['processes']]:
   if p.get('commandLine') is not None:
    text=p['commandLine'];key=sha(text.encode());dictionary[key]=text;p['commandLine']={'sha256':key}
  record['traceEncoding']={'kind':'LosslessCommandLineDictionary','version':'1.0.0','commandLines':dictionary}
  expanded=copy.deepcopy(record);encoding=expanded.pop('traceEncoding')
  for p in expanded['lifetimes']+[p for sample in expanded['observation']['samples'] for p in sample['processes']]:
   if isinstance(p.get('commandLine'),dict):p['commandLine']=encoding['commandLines'][p['commandLine']['sha256']]
  assert expanded==original
  put('native/'+name+'.json',record)
 return record
