def record(name,result,expected):
 target=EVIDENCE/(name+'.json');target.parent.mkdir(parents=True,exist_ok=True)
 result['id']=name;result['expectedExitCode']=expected;result['status']='PASS' if result['exitCode']==expected else 'FAIL'
 result['executionTimeHead']=subprocess.check_output(['git','rev-parse','HEAD'],cwd=ROOT,text=True).strip()
 result['harnessSha256']=sha(Path(__file__).read_bytes())
 result['distributionSha256']=sha((PKG/'distribution-manifest.json').read_bytes())
 target.write_bytes(j(result))
 cases.append(dict(id=name,status=result['status'],command=result['command'],exitCode=result['exitCode'],elapsedMs=result['elapsedMs'],evidencePaths=[target.relative_to(ROOT).as_posix()]))
 (EVIDENCE/'cases.json').write_bytes(j(cases))
 assert result['exitCode']==expected,(name,result['exitCode'],result['stdout'],result['stderr'])
 return result
def command(name,args,expected=0,env=ENV):
 return record(name,execute([str(NODE),'--max-old-space-size=128',*map(str,args)],str(ROOT),env=env,creationflags=subprocess.DETACHED_PROCESS|subprocess.CREATE_NEW_PROCESS_GROUP),expected)
def workspace(name,vector,change=None):
 dest=CACHE/'work'/'preliminary'/name
 dest.mkdir(parents=True,exist_ok=False)
 src=FIX/vector
 for filename in ['policy.json','candidate.mip','memoryos-ci.json']:
  shutil.copyfile(src/filename,dest/filename)
 config=json.loads((dest/'memoryos-ci.json').read_bytes())
 if change:change(config,dest)
 (dest/'memoryos-ci.json').write_bytes(j(config))
 return dest,config
def product(name,work,expected,extra=(),env=ENV):
 result=command(name,[PKG/'bin/memoryos-ci.mjs','run','--workspace',work,'--config',work/'memoryos-ci.json',*extra],expected,env)
 summary=json.loads(result['stdout']);assert summary['exitCode']==expected
 assert result['observation']['peakProcessCount']<=2
 assert result['observation']['peakAggregateRssBytes']<=768*1024*1024
 if summary['publication']=='COMPLETE':
  bundle=work/'.memoryos-ci/out'/summary['runId']
  target=EVIDENCE/'bundles'/name/summary['runId'];target.parent.mkdir(parents=True,exist_ok=True)
  shutil.copytree(bundle,target)
  command(name+'-verify',[PKG/'bin/memoryos-ci.mjs','verify','--bundle',bundle])
  return bundle
 return None

 for source,target in [('oracle-identity.json','evaluation-identity.json'),('oracle-outcome.json','policy-outcome.json')]:
  assert (work/source).read_bytes()==(bundle/target).read_bytes()
  if golden:assert (bundle/target).read_bytes()==(FIX/golden/target).read_bytes()
 args=[ROOT/'repositories/memoryos-cli/bin/memoryos.js','policy','evaluate','--policy-set' if config['operation']=='evaluatePolicySet' else '--policy',work/'policy.json','--package',work/'candidate.mip','--outcome',work/'cli-outcome.json','--identity-output',work/'cli-identity.json']
 if config['context'].get('baselineMip'):args+=['--regression-baseline',work/'baseline.mip']
 command(name+'-cli',args,EXIT[decision])
 assert (work/'cli-identity.json').read_bytes()==(bundle/'evaluation-identity.json').read_bytes()
 assert (work/'cli-outcome.json').read_bytes()==(bundle/'policy-outcome.json').read_bytes()
